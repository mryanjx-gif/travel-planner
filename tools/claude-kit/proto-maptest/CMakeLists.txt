cmake_minimum_required(VERSION 3.24)
project(MapTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.12 REQUIRED COMPONENTS Quick Gui GuiPrivate ShaderTools)
find_package(QMapLibre REQUIRED COMPONENTS Quick)
qt_standard_project_setup(REQUIRES 6.12)
qt_add_executable(maptest main.cpp triangleitem.cpp triangleitem.h)
qt_add_qml_module(maptest URI MapTestApp VERSION 1.0 QML_FILES Main.qml)
qt_add_shaders(maptest "tri_shaders" PREFIX "/" FILES shaders/tri.vert shaders/tri.frag)
target_link_libraries(maptest PRIVATE Qt6::Quick Qt6::GuiPrivate)
